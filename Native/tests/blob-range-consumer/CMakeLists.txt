cmake_minimum_required(VERSION 3.31)
project(InstalledBlobRangeConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 ON)
find_package(iiFileProviderNative 0.11 CONFIG REQUIRED)
if(TARGET Qt6::Core OR TARGET Qt6::Gui OR TARGET Qt5::Core)
    message(FATAL_ERROR "Native BLOB range consumer imported Qt")
endif()
add_executable(ranges ../database_ranges_test.cpp)
target_link_libraries(ranges PRIVATE iiFileProvider::Native)
add_executable(allocation ../database_ranges_allocation_test.cpp)
target_link_libraries(allocation PRIVATE iiFileProvider::Native)
add_executable(database ../database_test.cpp)
target_link_libraries(database PRIVATE iiFileProvider::Native)
enable_testing()
add_test(NAME InstalledBlobRanges.Contract COMMAND ranges "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME InstalledBlobRanges.EightK COMMAND ranges "${CMAKE_CURRENT_BINARY_DIR}" --large)
add_test(NAME InstalledBlobRanges.Allocation COMMAND allocation)
add_test(NAME InstalledBlobRanges.Database COMMAND database "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(InstalledBlobRanges.EightK PROPERTIES TIMEOUT 180)
get_property(_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${_tests} PROPERTIES ENVIRONMENT_MODIFICATION
    "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_VERSIONED_LIBRARY_PATH=unset:;LD_LIBRARY_PATH=unset:")
