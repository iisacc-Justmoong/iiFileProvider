cmake_minimum_required(VERSION 3.31)
project(iiFileProviderNative VERSION 0.8.1 LANGUAGES C CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(GenerateExportHeader)
include(CTest)
include(../cmake/iiFileProviderSQLite.cmake)
include(VerifyDependencies.cmake)
option(IIFILEPROVIDER_NATIVE_SANITIZERS "Instrument the standard native component" OFF)
if(IIFILEPROVIDER_NATIVE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "Native sanitizer validation requires Clang or GCC")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(_native_kind SHARED)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(_native_kind STATIC)
endif()
add_library(iiFileProviderNative ${_native_kind} src/Storage.cpp src/FileProviderC.cpp src/Authorship.cpp src/File.cpp src/Sha256.cpp src/Presentation.cpp src/PresentationImage.cpp ../src/TemporaryLease.cpp third_party/utf8proc/utf8proc.c)
add_library(iiFileProvider::Native ALIAS iiFileProviderNative)
target_link_libraries(iiFileProviderNative PRIVATE SQLite::SQLite3 Threads::Threads)
if(ANDROID)
    target_sources(iiFileProviderNative PRIVATE src/PresentationAndroid.cpp)
endif()
if(APPLE)
    enable_language(OBJCXX)
    set_target_properties(iiFileProviderNative PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO)
    target_sources(iiFileProviderNative PRIVATE src/PresentationApple.mm)
    set_source_files_properties(src/PresentationApple.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(iiFileProviderNative PRIVATE "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics" "-framework QuickLookThumbnailing")
endif()
if(WIN32)
    target_link_libraries(iiFileProviderNative PRIVATE ntdll)
endif()
set_target_properties(iiFileProviderNative PROPERTIES EXPORT_NAME Native EXPORT_NO_SYSTEM TRUE CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden C_VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION 0.5)
target_compile_features(iiFileProviderNative PUBLIC cxx_std_23)
if(EMSCRIPTEN)
    target_compile_options(iiFileProviderNative PUBLIC -pthread)
    target_compile_options(iiFileProviderNative PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:-fexperimental-library>")
    target_link_options(iiFileProviderNative PUBLIC -pthread)
endif()
target_compile_definitions(iiFileProviderNative PRIVATE UTF8PROC_STATIC)
target_include_directories(iiFileProviderNative PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE third_party third_party/utf8proc ../src)
generate_export_header(iiFileProviderNative EXPORT_MACRO_NAME IIFILEPROVIDER_NATIVE_EXPORT EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/generated/iiFileProviderNativeExport.h")
if(NOT MSVC)
 target_compile_options(iiFileProviderNative PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS iiFileProviderNative EXPORT iiFileProviderNativeTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/iiFileProviderPresentation.h src/iiFileProviderStorage.h src/FileProviderC.h src/iiFileProviderNative.h "${CMAKE_CURRENT_BINARY_DIR}/generated/iiFileProviderNativeExport.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT iiFileProviderNativeTargets NAMESPACE iiFileProvider:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderNative)
configure_package_config_file(iiFileProviderNativeConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderNativeConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderNative)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderNativeConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderNativeConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderNativeConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderNative)
install(DIRECTORY third_party/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiFileProviderNative)
install(FILES third_party/utf8proc/LICENSE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiFileProviderNative/utf8proc)
install(FILES third_party/stb/LICENSE third_party/stb/provenance.json DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiFileProviderNative/stb)
install(FILES ../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiFileProviderNative)
install(FILES third_party/dependencies.json DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiFileProviderNative)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/iiFileProviderNative)
install(FILES ../docs/FILE_PRESENTATION.md ../docs/FILE_PRESENTATION_VALIDATION.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/iiFileProviderNative)
install(DIRECTORY ../python/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiFileProvider/python
    FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
if(BUILD_TESTING)
 add_executable(iiFileProviderPresentationTest tests/presentation_test.cpp)
 target_link_libraries(iiFileProviderPresentationTest PRIVATE iiFileProvider::Native)
 if(WIN32)
  target_link_libraries(iiFileProviderPresentationTest PRIVATE advapi32)
 endif()
 target_include_directories(iiFileProviderPresentationTest PRIVATE third_party)
 add_test(NAME iiFileProvider.Native.Presentation COMMAND iiFileProviderPresentationTest "${CMAKE_CURRENT_BINARY_DIR}")
 set_tests_properties(iiFileProvider.Native.Presentation PROPERTIES TIMEOUT 90)
 add_executable(iiFileProviderNativeTest tests/native_test.cpp)
 target_link_libraries(iiFileProviderNativeTest PRIVATE iiFileProvider::Native)
 add_test(NAME iiFileProvider.Native COMMAND iiFileProviderNativeTest)
 add_executable(iiFileProviderNativeAuthorLinksTest tests/author_links_test.cpp)
 target_link_libraries(iiFileProviderNativeAuthorLinksTest PRIVATE iiFileProvider::Native)
 target_include_directories(iiFileProviderNativeAuthorLinksTest PRIVATE third_party)
 add_test(NAME iiFileProvider.Native.AuthorLinks COMMAND iiFileProviderNativeAuthorLinksTest)
endif()
