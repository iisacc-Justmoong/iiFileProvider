 elseif(APPLE)
  set_property(TARGET iiFileProviderNative APPEND PROPERTY INSTALL_RPATH "@loader_path")

  set_property(TARGET iiFileProviderNative APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
 endif()

  target_compile_options(iiFileProviderNative PRIVATE -fPIC -sSIDE_MODULE=1)
  target_link_options(iiFileProviderNative PRIVATE -sSIDE_MODULE=1)
 endif()
endif()

  endif()
 endif()
endif()
