cmake_minimum_required(VERSION 3.24)
project(iiFileProviderObjectsInstalled LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(iiFileProviderObjects CONFIG REQUIRED)
if(TARGET Qt6::Core)
    message(FATAL_ERROR "The object package unexpectedly loaded Qt")
endif()
add_executable(objects_installed main.cpp)
target_link_libraries(objects_installed PRIVATE iiFileProvider::Objects)
enable_testing()
add_test(NAME objects.installed COMMAND objects_installed)
foreach(_test IN ITEMS object_document object_journal object_journal_storage object_metadata_limits ontology_filesystem ontology_storage ontology_boundaries)
    add_executable(installed_${_test} ../${_test}.cpp)
    target_compile_definitions(installed_${_test} PRIVATE OBJECT_TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(installed_${_test} PRIVATE iiFileProvider::Objects SQLite3::SQLite3 Threads::Threads)
    add_test(NAME objects.installed.${_test} COMMAND installed_${_test})
    set_tests_properties(objects.installed.${_test} PROPERTIES TIMEOUT 120)
endforeach()
