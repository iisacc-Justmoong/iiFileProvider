cmake_minimum_required(VERSION 3.24)
project(iiFileProviderCatalogInstalled LANGUAGES CXX)
find_package(iiFileProviderCatalog 0.6.0 CONFIG REQUIRED)
if(TARGET Qt6::Core OR TARGET Qt5::Core)
    message(FATAL_ERROR "The native catalog package unexpectedly loaded Qt")
endif()
add_executable(catalog_installed ../file_catalog.cpp)
target_compile_definitions(catalog_installed PRIVATE CATALOG_TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(catalog_installed PRIVATE iiFileProvider::Catalog)
enable_testing()
add_test(NAME catalog.installed COMMAND catalog_installed)
