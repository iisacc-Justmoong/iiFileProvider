cmake_minimum_required(VERSION 3.31)
project(FileOwnershipConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 TRUE)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED)
if(TARGET Qt6::Core)
    message(FATAL_ERROR "Native storage imported Qt")
endif()
add_executable(ownership_consumer main.cpp)
target_link_libraries(ownership_consumer PRIVATE iiFileProvider::Native)
enable_testing()
add_test(NAME FileOwnership.NativeInstalled COMMAND ownership_consumer ${CMAKE_CURRENT_BINARY_DIR})
