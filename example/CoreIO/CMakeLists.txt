cmake_minimum_required(VERSION 3.31)
# iisdk:consumer-application
project(iiFileProviderCoreIO VERSION 0.1.1 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(iiSDKCore 0.13 CONFIG REQUIRED COMPONENTS Runtime)
if(NOT TARGET iiFileProvider::Native)
    find_package(iiFileProviderNative 0.7 CONFIG REQUIRED)
endif()
add_library(iiFileProviderCoreIO STATIC App/Channel.cpp App/AtomicFile.cpp)
add_library(iiFileProvider::CoreIO ALIAS iiFileProviderCoreIO)
set_target_properties(iiFileProviderCoreIO PROPERTIES EXPORT_NAME CoreIO CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_features(iiFileProviderCoreIO PUBLIC cxx_std_23)
target_include_directories(iiFileProviderCoreIO PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/App>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(iiFileProviderCoreIO PUBLIC iiSDKCore::iiSDKCore iiFileProvider::Native)
if(NOT MSVC)
    target_compile_options(iiFileProviderCoreIO PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
install(TARGETS iiFileProviderCoreIO EXPORT iiFileProviderCoreIOTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES App/iiFileProviderCoreIO.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT iiFileProviderCoreIOTargets NAMESPACE iiFileProvider:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderCoreIO)
configure_package_config_file(iiFileProviderCoreIOConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCoreIOConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderCoreIO)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCoreIOConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCoreIOConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCoreIOConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderCoreIO)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiFileProviderCoreIO)
add_executable(iiFileProviderCoreIOExample examples/loop_io.cpp)
target_link_libraries(iiFileProviderCoreIOExample PRIVATE iiFileProvider::CoreIO)
if(BUILD_TESTING)
    add_executable(iiFileProviderCoreIOTests tests/contracts.cpp)
    target_link_libraries(iiFileProviderCoreIOTests PRIVATE iiFileProvider::CoreIO)
    add_test(NAME iiFileProvider.CoreIO COMMAND iiFileProviderCoreIOTests)
    add_test(NAME iiFileProvider.CoreIOExample COMMAND iiFileProviderCoreIOExample)
    set_tests_properties(iiFileProvider.CoreIO iiFileProvider.CoreIOExample PROPERTIES TIMEOUT 90
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources")
endif()
