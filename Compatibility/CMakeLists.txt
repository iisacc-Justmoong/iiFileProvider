cmake_minimum_required(VERSION 3.31)
project(iiFileProviderCompatibility VERSION 0.9.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(BUILD_SHARED_LIBS "Build file compatibility as a shared library" ON)
if(EMSCRIPTEN AND BUILD_SHARED_LIBS)
    set_property(GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS TRUE)
endif()
if(NOT TARGET iiFileProvider::Native)
    find_package(iiFileProviderNative 0.9 CONFIG REQUIRED)
endif()
add_library(iiFileProviderCompatibility src/Compatibility.cpp)
add_library(iiFileProvider::Compatibility ALIAS iiFileProviderCompatibility)
target_compile_features(iiFileProviderCompatibility PUBLIC cxx_std_23)
target_include_directories(iiFileProviderCompatibility PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(iiFileProviderCompatibility PUBLIC iiFileProvider::Native)
set_target_properties(iiFileProviderCompatibility PROPERTIES
        AUTOMOC OFF AUTORCC OFF AUTOUIC OFF POSITION_INDEPENDENT_CODE ON)
set_target_properties(iiFileProviderCompatibility PROPERTIES EXPORT_NAME Compatibility CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(BUILD_SHARED_LIBS)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set_target_properties(iiFileProviderCompatibility PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER "com.iisacc.iiFileProviderCompatibility")
    elseif(APPLE)
        set_property(TARGET iiFileProviderCompatibility APPEND PROPERTY INSTALL_RPATH "@loader_path")
    elseif(UNIX AND NOT EMSCRIPTEN)
        set_property(TARGET iiFileProviderCompatibility APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
    endif()
    if(EMSCRIPTEN)
        target_compile_options(iiFileProviderCompatibility PRIVATE -fPIC -sSIDE_MODULE=1)
        target_link_options(iiFileProviderCompatibility PRIVATE -sSIDE_MODULE=1)
    endif()
endif()
set(_package ${CMAKE_INSTALL_LIBDIR}/cmake/iiFileProviderCompatibility)
install(TARGETS iiFileProviderCompatibility EXPORT iiFileProviderCompatibilityTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} FRAMEWORK DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.h")
install(EXPORT iiFileProviderCompatibilityTargets NAMESPACE iiFileProvider:: DESTINATION ${_package})
configure_package_config_file(iiFileProviderCompatibilityConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCompatibilityConfig.cmake" INSTALL_DESTINATION ${_package})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCompatibilityConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCompatibilityConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiFileProviderCompatibilityConfigVersion.cmake" DESTINATION ${_package})
include(CTest)
if(BUILD_TESTING)
    add_executable(iiFileProviderCompatibilityContracts tests/contracts.cpp)
    target_link_libraries(iiFileProviderCompatibilityContracts PRIVATE iiFileProvider::Compatibility)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    include(../cmake/TestWorkspace.cmake)
    set(_iifp_test_runner "${CMAKE_COMMAND}" -E env "IIFILEPROVIDER_NATIVE_LIBRARY=$<TARGET_FILE:iiFileProvider::Native>"
        "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../tests/resource_runner.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/../python" "${CMAKE_CURRENT_SOURCE_DIR}/../resources")
    iifp_fixture_test(iiFileProvider.Compatibility.Contracts iiFileProviderCompatibilityContracts @WORK@)
    if(BUILD_SHARED_LIBS AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME iiFileProvider.Compatibility.SharedType COMMAND "${CMAKE_COMMAND}"
            "-DTARGET_TYPE=$<TARGET_PROPERTY:iiFileProviderCompatibility,TYPE>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared_target_type.cmake")
        add_test(NAME iiFileProvider.Compatibility.SharedLink COMMAND "${CMAKE_COMMAND}"
            "-DIMAGE=$<TARGET_FILE:iiFileProviderCompatibilityContracts>"
            "-DEXPECTED=$<TARGET_FILE:iiFileProviderCompatibility>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/../tests/shared_link.cmake")
    endif()
endif()
